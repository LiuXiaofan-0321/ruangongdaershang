# 第6讲作业一和作业二：可信度量模型 T 与 T₃

## 一、作业一：计算模型 T

### 1. 题目与数据

软件有五个属性：功能性 $F$、安全性 $SF$、可靠性 $R$、生存性 $SV$ 和可维护性 $M$。按照课件中不区分关键属性和非关键属性的乘积模型计算软件可信性度量值。

| 属性 | $F$ | $SF$ | $R$ | $SV$ | $M$ |
|---|---:|---:|---:|---:|---:|
| 权重 | 0.25 | 0.15 | 0.20 | 0.23 | 0.17 |

模型及本题代入式为：

$$
T=\prod_{i=1}^{m}y_i^{\alpha_i}
=F^{0.25}SF^{0.15}R^{0.20}SV^{0.23}M^{0.17}。
$$

例如第一组数据：

$$
T=6.8^{0.25}\times 8.2^{0.15}\times 7.8^{0.20}
\times 6.8^{0.23}\times 7.9^{0.17}=7.373809。
$$

## 二、作业二：计算模型 T₃

### 1. 题目与方法

四个软件系统各有七个属性。$y_1,y_2,y_3,y_4$ 为关键属性，权重分别为 $(0.30,0.20,0.35,0.15)$；$y_5,y_6,y_7$ 为非关键属性，权重分别为 $(0.35,0.40,0.25)$。关键属性组权重 $\alpha=0.70$，非关键属性组权重 $\beta=0.30$。

按课件中的模型：

$$
T_3=\left\{\alpha\left[\min_{1\le i\le4}\left(\frac{y_i}{10}\right)^{\varepsilon}
\prod_{i=1}^{4}y_i^{\alpha_i}\right]^{-\rho}
+\beta\left[\prod_{j=5}^{7}y_j^{\beta_{j-4}}\right]^{-\rho}\right\}^{-1/\rho}。
$$

先计算关键属性的加权乘积，再乘以最小惩罚因子；再计算非关键属性的加权乘积，最后按上式合成。以系统 1 的第一行为例：

$$
\min_{1\le i\le4}(y_i/10)^{0.1}=0.963574,
\quad \prod_{i=1}^{4}y_i^{\alpha_i}=7.758401,
$$

$$
A=0.963574\times7.758401\approx7.475791,\quad
B=9.1^{0.35}\times8.5^{0.40}\times8.6^{0.25}\approx8.730853,
$$

$$
T_3=(0.70A^{-0.10}+0.30B^{-0.10})^{-10}=7.830101。
$$

上式展示的中间值已四舍五入，程序使用未截断的中间值。

## 三、C++ 程序

程序将题目数据直接写入数组，分别调用 `modelT` 和 `modelT3`，输出保留六位小数。编译与运行：

```powershell
g++ -std=c++17 -Wall -Wextra -pedantic 第6讲作业一二.cpp -o 第6讲作业一二.exe
.\第6讲作业一二.exe
```

```cpp
#include <algorithm>
#include <array>
#include <cmath>
#include <iomanip>
#include <iostream>

using namespace std;

// 作业一：不区分关键和非关键属性的乘积模型。
double modelT(const array<double, 5>& y, const array<double, 5>& weight) {
    double result = 1.0;
    for (size_t i = 0; i < y.size(); ++i) {
        result *= pow(y[i], weight[i]);
    }
    return result;
}

// 作业二：前 4 项为关键属性，后 3 项为非关键属性。
double modelT3(const array<double, 7>& y, double epsilon, double rho) {
    const array<double, 4> criticalWeight = {0.30, 0.20, 0.35, 0.15};
    const array<double, 3> noncriticalWeight = {0.35, 0.40, 0.25};
    const double alpha = 0.70, beta = 0.30;

    double minFactor = pow(y[0] / 10.0, epsilon);
    double criticalProduct = 1.0, noncriticalProduct = 1.0;
    for (size_t i = 0; i < criticalWeight.size(); ++i) {
        minFactor = min(minFactor, pow(y[i] / 10.0, epsilon));
        criticalProduct *= pow(y[i], criticalWeight[i]);
    }
    for (size_t i = 0; i < noncriticalWeight.size(); ++i) {
        noncriticalProduct *= pow(y[i + 4], noncriticalWeight[i]);
    }
    double criticalPart = minFactor * criticalProduct;
    return pow(alpha * pow(criticalPart, -rho)
             + beta * pow(noncriticalProduct, -rho), -1.0 / rho);
}

int main() {
    const array<double, 5> tWeight = {0.25, 0.15, 0.20, 0.23, 0.17};
    const array<array<double, 5>, 4> tData = {{
        {{6.8, 8.2, 7.8, 6.8, 7.9}},
        {{8.8, 8.9, 9.2, 9.0, 9.3}},
        {{5.6, 5.9, 6.0, 9.0, 5.3}},
        {{4.6, 7.8, 3.8, 5.1, 4.5}}
    }};
    struct Case {
        int system;
        array<double, 7> y;
        double epsilon, rho;
    };
    const array<Case, 8> t3Data = {{
        {1, {{8.3, 7.1, 8.1, 6.9, 9.1, 8.5, 8.6}}, 0.10, 0.10},
        {1, {{7.8, 8.7, 8.0, 6.2, 8.9, 8.0, 8.8}}, 0.05, 0.15},
        {2, {{5.8, 8.1, 6.8, 5.6, 7.9, 6.8, 8.1}}, 0.10, 0.10},
        {2, {{6.8, 7.8, 8.1, 6.9, 8.9, 7.8, 8.8}}, 0.05, 0.15},
        {3, {{3.8, 3.7, 4.8, 4.7, 3.9, 5.8, 6.8}}, 0.10, 0.10},
        {3, {{8.4, 7.2, 8.1, 7.7, 7.9, 0.88, 0.68}}, 0.05, 0.15},
        {4, {{9.0, 8.7, 8.9, 8.6, 9.1, 8.8, 8.9}}, 0.10, 0.10},
        {4, {{8.0, 7.8, 6.8, 8.7, 7.9, 8.1, 8.3}}, 0.05, 0.15}
    }};

    cout << fixed << setprecision(6);
    cout << "Homework 1: T\n";
    for (size_t i = 0; i < tData.size(); ++i)
        cout << "Row " << i + 1 << ": " << modelT(tData[i], tWeight) << '\n';
    cout << "\nHomework 2: T3\n";
    for (size_t i = 0; i < t3Data.size(); ++i) {
        const auto& c = t3Data[i];
        cout << "System " << c.system << ", row " << i % 2 + 1
             << ": " << modelT3(c.y, c.epsilon, c.rho) << '\n';
    }
    return 0;
}
